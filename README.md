## Inline code example
#include <stdio.h>

// The 'inline' keyword suggests the compiler substitute the code directly
inline int square(int num) {
    return num * num;
}

int main() {
    int val = 5;
    printf("The square of %d is %d\n", val, square(val));
    
    return 0;
}
